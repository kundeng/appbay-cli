## How to run the journeys on the Lima guests

```bash
cd apps/cli && bun build --compile --target=bun-linux-arm64 src/index.ts --outfile dist/appbay-linux-arm64 && cd ../..
mkdir -p /tmp/mp && ln -sf "$PWD/scripts/journeys/lib/multipass-lima-shim" /tmp/mp/multipass

# Podman guest: rootful Podman 5.8, real install at /root/appbay-pv
limactl copy apps/cli/dist/appbay-linux-arm64 podman:/tmp/appbay.new
limactl shell --workdir /tmp podman sudo bash -c 'install -m 755 /tmp/appbay.new /usr/local/bin/appbay && rm -f /tmp/appbay.new'
PATH=/tmp/mp:$PATH VM=podman PRIV=sudo CBIN=podman HOME_DIR=/root/appbay-pv WORKDIR=/root \
  ./scripts/journeys/s29-journey-deploy-reporting.sh   # then s26-journey-apply-success, s26-journey-lifecycle

# Rocky guest: Docker 29, real install at /var/lib/appbay; sudo drops /usr/local/bin from PATH
limactl copy apps/cli/dist/appbay-linux-arm64 rocky:/tmp/appbay.new
limactl shell --workdir /tmp rocky sudo bash -c 'install -m 755 /tmp/appbay.new /usr/local/bin/appbay && rm -f /tmp/appbay.new'
PATH=/tmp/mp:$PATH VM=rocky PRIV="sudo env PATH=/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin" CBIN=docker HOME_DIR=/var/lib/appbay WORKDIR=/root \
  ./scripts/journeys/s29-journey-deploy-reporting.sh
```
